    static double factorial(double v)
    {
        if (v < 0 || v != std::floor(v) || v > 170) fail("factorial needs an integer from 0 to 170");
        double r = 1;
        for (int i = 2; i <= static_cast<int>(v); ++i) r *= i;
        return r;
    }
